Usage

Apply a theme by setting colors before creating your window:

lua 
local KetamineUI = loadstring(game:HttpGet("YOUR_LINK"))()

-- Apply the "Neon" theme
KetamineUI.Theme.Accent    = Color3.fromRGB(0, 255, 170)
KetamineUI.Theme.AccentDim = Color3.fromRGB(0, 180, 120)
KetamineUI.Theme.BG        = Color3.fromRGB(5, 5, 8)
KetamineUI.Theme.BG2       = Color3.fromRGB(10, 10, 16)
KetamineUI.Theme.BG3       = Color3.fromRGB(17, 17, 25)
KetamineUI.Theme.Text      = Color3.fromRGB(0, 255, 170)
KetamineUI.Theme.TextDim   = Color3.fromRGB(0, 140, 100)
KetamineUI.Theme.On        = Color3.fromRGB(0, 255, 170)
KetamineUI.Theme.Off       = Color3.fromRGB(20, 20, 30)

local Window = KetamineUI:CreateWindow({Name = "Neon Hub"})
