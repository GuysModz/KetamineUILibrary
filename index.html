Installation

Load KetamineUI in your script with a single line. Host the library file on GitHub or any raw file host.

lua 
local KetamineUI = loadstring(game:HttpGet("https://raw.githubusercontent.com/GuysModz/Ketamine-UI-Library/refs/heads/main/ketamineUI.lua"))()

Or load from a local file:

lua 
local KetamineUI = loadstring(readfile("KetamineUI.lua"))()

Quick Start

Create a window, add a tab, and start building your UI in seconds.

lua 
local KetamineUI = loadstring(game:HttpGet("https://raw.githubusercontent.com/GuysModz/Ketamine-UI-Library/refs/heads/main/ketamineUI.lua"))()

local Window = KetamineUI:CreateWindow({
    Name = "My Script",
    Subtitle = "v1.0"
})

local MainTab = Window:CreateTab("Main")

MainTab:CreateToggle({
    Name = "Auto Farm",
    Default = false,
    Callback = function(value)
        getgenv().AutoFarm = value
    end
})

Creating a Window

The window is the main container. It includes a draggable title bar, close/minimize buttons, and houses all your tabs.

lua 
local Window = KetamineUI:CreateWindow({
    Name = "My Hub",                          -- Window title
    Subtitle = "v2.1",                        -- Subtitle (optional)
    Size = UDim2.new(0, 400, 0, 520),         -- Window size (optional)
    ToggleKey = Enum.KeyCode.RightControl     -- Toggle keybind (optional)
})

Parameters

ParameterTypeDefaultDescription
Namestring"Ketamine UI"Title displayed in title bar
Subtitlestring""Small text below title
SizeUDim2380×500Window dimensions
ToggleKeyEnum.KeyCodeRightControlKey to show/hide UI

Tabs

Tabs organize your features into separate pages. A tab bar renders below the title. The first tab is automatically selected.

lua 
local CombatTab = Window:CreateTab("Combat")
local VisualsTab = Window:CreateTab("Visuals")
local MiscTab = Window:CreateTab("Misc")
💡 You can switch tabs programmatically with Window:SwitchTab("TabName")
⚠️ Important: Use the variable you stored the tab in (e.g. CombatTab:CreateToggle), NOT just Tab:. Using an undefined variable will silently fail and no elements will appear.

Theme Customization

Override theme colors before creating your window to match your brand.

lua 
local KetamineUI = loadstring(game:HttpGet("YOUR_LINK"))()

-- Customize to red theme

Section

A styled header to visually group related elements.

lua 
CombatTab:CreateSection("AIMBOT")

Toggle

An animated on/off switch. Returns an object with :Set() and :Get() methods for programmatic control.

lua 
local MyToggle = CombatTab:CreateToggle({
    Name = "Enable Aimbot",
    Default = false,
    Callback = function(Value)
        print("Aimbot:", Value)
    end
})

-- Programmatic control:
MyToggle:Set(true)
print(MyToggle:Get()) -- true

Parameters

ParameterTypeDefaultDescription
Namestring"Toggle"Display name
DefaultbooleanfalseInitial state
CallbackfunctionnilCalled with true/false

Slider

A draggable slider for numeric values. Displays the current value and calls back on change.

lua 
local FOV = CombatTab:CreateSlider({
    Name = "FOV Radius",
    Min = 50,
    Max = 800,
    Default = 200,
    Callback = function(Value)
        getgenv().AimbotFOV = Value
    end
})

-- Update programmatically:
FOV:Set(400)

Parameters

ParameterTypeDefaultDescription
Namestring"Slider"Display name
Minnumber0Minimum value
Maxnumber100Maximum value
DefaultnumberMinInitial value
CallbackfunctionnilCalled with the number value

Button

A clickable button with hover animation and press feedback.

lua 
MiscTab:CreateButton({
    Name = "Reset Character",
    Callback = function()
        game.Players.LocalPlayer.Character:BreakJoints()
    end
})

Parameters

ParameterTypeDefaultDescription
Namestring"Button"Button text
CallbackfunctionnilCalled on click

Dropdown

A selection picker that expands to show options. Supports dynamic refresh.

lua 
local PartPicker = CombatTab:CreateDropdown({
    Name = "Target Part",
    Options = {"Head", "HumanoidRootPart", "UpperTorso"},
    Default = "Head",
    Callback = function(Value)
        getgenv().TargetPart = Value
    end
})

-- Change selection:
PartPicker:Set("UpperTorso")

-- Update options dynamically:
PartPicker:Refresh({"Head", "Torso", "LeftArm", "RightArm"})

Parameters

ParameterTypeDefaultDescription
Namestring"Dropdown"Display label
Optionstable{}Array of string options
DefaultstringFirst optionInitially selected
CallbackfunctionnilCalled with selected string

Textbox

A text input field. Callback fires when the user presses Enter.

lua 
local Input = MiscTab:CreateTextbox({
    Name = "Player Name",
    Default = "",
    Placeholder = "Enter username...",
    Callback = function(Value)
        print("Searching for:", Value)
    end
})

-- Set programmatically:
Input:Set("Player123")

Parameters

ParameterTypeDefaultDescription
Namestring"Input"Display label
Defaultstring""Initial text
Placeholderstring"..."Placeholder when empty
CallbackfunctionnilCalled on Enter press

Label

A read-only text display for status info. Update dynamically with :Set().

lua 
local Status = MiscTab:CreateLabel("Status: Idle")

-- Update later:
Status:Set("Status: Farming...")
Status:Set("FPS: " .. fps)

Notifications

Send native Roblox notification popups.

lua 
KetamineUI:Notify({
    Title = "Success",
    Text = "Script loaded successfully!",
    Duration = 3
})

Parameters

ParameterTypeDefaultDescription

Methods

lua 
-- Destroy the window
Window:Destroy()

-- Destroy ALL KetamineUI instances
KetamineUI:Destroy()

-- Switch to a specific tab
Window:SwitchTab("Visuals")

-- Element methods (Toggle, Slider, Dropdown, Textbox, Label)
element:Set(value)    -- Set value programmatically
element:Get()         -- Get current value

-- Dropdown-specific
dropdown:Refresh({"new", "options"})

Full Example

A complete script demonstrating all elements:

lua 
local KetamineUI = loadstring(game:HttpGet("YOUR_RAW_LINK"))()

local Window = KetamineUI:CreateWindow({
    Name = "Example Hub",
    Subtitle = "v1.0 | Full Demo"
})

-- Combat Tab
local Combat = Window:CreateTab("Combat")
Combat:CreateSection("AIMBOT")

local aimbot = Combat:CreateToggle({
    Name = "Enable Aimbot",
    Default = false,
    Callback = function(v) getgenv().Aimbot = v end
})

Combat:CreateSlider({
    Name = "FOV Radius",
    Min = 50, Max = 800, Default = 200,
    Callback = function(v) getgenv().FOV = v end
})

Combat:CreateDropdown({
    Name = "Target Part",
    Options = {"Head", "HumanoidRootPart", "UpperTorso"},
    Default = "Head",
    Callback = function(v) getgenv().Part = v end
})

-- Visuals Tab
local Visuals = Window:CreateTab("Visuals")
Visuals:CreateSection("ESP")

Visuals:CreateToggle({
    Name = "Player ESP",
    Default = false,
    Callback = function(v) getgenv().ESP = v end
})

Visuals:CreateToggle({
    Name = "Fullbright",
    Default = false,
    Callback = function(v)
        game.Lighting.Brightness = v and 2 or 1
        game.Lighting.FogEnd = v and 100000 or 1000
    end
})

-- Misc Tab
local Misc = Window:CreateTab("Misc")
Misc:CreateSection("PLAYER")

Misc:CreateSlider({
    Name = "Walk Speed",
    Min = 16, Max = 200, Default = 16,
    Callback = function(v)
        local hum = game.Players.LocalPlayer.Character
            and game.Players.LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if hum then hum.WalkSpeed = v end
    end
})

Misc:CreateTextbox({
    Name = "TP to Player",
    Placeholder = "Username...",
    Callback = function(v)
        local t = game.Players:FindFirstChild(v)
        if t and t.Character then
            game.Players.LocalPlayer.Character:MoveTo(
                t.Character.HumanoidRootPart.Position
            )
        end
    end
})

Misc:CreateButton({
    Name = "Reset Character",
    Callback = function()
        game.Players.LocalPlayer.Character:BreakJoints()
    end
})

Misc:CreateSection("INFO")
local fpsLabel = Misc:CreateLabel("FPS: --")

task.spawn(function()
    while task.wait(1) do
        local fps = math.floor(1 / game:GetService("RunService").RenderStepped:Wait())
        fpsLabel:Set("FPS: " .. fps)
    end
end)

KetamineUI:Notify({
    Title = "Example Hub",
    Text = "Loaded! RightCtrl to toggle.",
    Duration = 4
})
